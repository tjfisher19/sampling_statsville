function normalPDF(x, mu, sigma) {
  return (1 / (sigma * Math.sqrt(2 * Math.PI))) * Math.exp(-0.5 * Math.pow((x - mu) / sigma, 2));
}

// Error Function (erf) approximation for Cumulative Distribution Function (CDF)
function erf(x) {
  const a1 =  0.254829592;
  const a2 = -0.284496736;
  const a3 =  1.421413741;
  const a4 = -1.453152027;
  const a5 =  1.061405429;
  const p  =  0.3275911;

  const sign = (x < 0) ? -1 : 1;
  const absX = Math.abs(x);

  const t = 1.0 / (1.0 + p * absX);
  const y = 1.0 - (((((a5 * t + a4) * t) + a3) * t + a2) * t + a1) * t * Math.exp(-absX * absX);

  return sign * y;
}

// Normal Cumulative Distribution Function (CDF)
function normalCDF(x, mu, sigma) {
  return 0.5 * (1 + erf((x - mu) / (sigma * Math.sqrt(2))));
}

// Generate coordinate points to draw the curve smoothly
curveData = {
  const points = [];
  const minX = mean - 4 * sd;
  const maxX = mean + 4 * sd;
  const steps = 200;
  const stepSize = (maxX - minX) / steps;
  
  for (let i = 0; i <= steps; i++) {
    let currentX = minX + i * stepSize;
    points.push({ x: currentX, y: normalPDF(currentX, mean, sd) });
  }
  return points;
}

// Calculate the final probability output dynamically
probability = {
  const pBelow = normalCDF(x_val, mean, sd);
  return direction === "Below (≤)" ? pBelow : 1 - pBelow;
}
